// Copie este archivo a src/main.cpp y configure la red antes de compilar.
// Conserve include/MicroROSHelper.h y el transporte wifi en platformio.ini.
#include <Arduino.h>
#include "MicroROSHelper.h"
#include <std_msgs/msg/bool.h>

const size_t ROS_DOMAIN = 13;
const size_t EXECUTOR_HANDLES = 1;
char WIFI_SSID[] = "NOMBRE_RED";
char WIFI_PASSWORD[] = "CLAVE_RED";
IPAddress AGENT_IP(192, 168, 1, 100);
const uint16_t AGENT_PORT = 8888;
const char NODE_NAME[] = "ejemplo_subscriber";
const uint8_t LED_PIN = 2; // LED activo en alto; adapte el pin a su placa.

// Un subscriber consume un handle. No se necesita un timer.
MicroROSHelper ros;
std_msgs__msg__Bool led_msg;
rcl_subscription_t led_sub;

void led_callback(const void * msgin)
{
    if (msgin == nullptr) return;
    const std_msgs__msg__Bool * msg = (const std_msgs__msg__Bool *) msgin;
    digitalWrite(LED_PIN, msg->data ? HIGH : LOW);
}

void setup()
{
    pinMode(LED_PIN, OUTPUT);
    digitalWrite(LED_PIN, LOW);
    bool ok = ros.beginWiFi(
        NODE_NAME, EXECUTOR_HANDLES, ROS_DOMAIN,
        WIFI_SSID, WIFI_PASSWORD, AGENT_IP, AGENT_PORT
    );
    ros.require(ok, "Error inicializando micro-ROS");
    ok = ros.createSubscription(
        led_sub,
        ROSIDL_GET_MSG_TYPE_SUPPORT(std_msgs, msg, Bool),
        "/led", &led_msg, led_callback
    );
    ros.require(ok, "Error creando subscriber /led");
    ros.info("Nodo listo");
}

void loop()
{
    if (!ros.wifiConnected()) {
        delay(1000);
        return;
    }
    ros.spinSome(10);
    delay(10);
}

/*
